#表子查询
#1.查询与"杨逍" , "张无忌"的职位和薪资相同的员工信息
#a.查询"杨逍" ， “张无忌" 的职位和薪资

select job,salary from emp where name = '杨逍' or name = '张无忌';

#b.查询与"鹿杖客" , "宋远桥" 的职位和薪资相同的员工信息
select * from emp where (job,salary) in (select job,salary from emp where name = '杨逍' or name = '张无忌');

#.查询入职日期是"2002-11-03"之后的员工信息,及其部门信息
#a.入职日期是"2002-11-03"之后的员工信息
select * from emp where entrydate > '2002-11-03';

#b.查询这部分员工，对应的部门信息;
select * from(select * from emp where entrydate > '2002-11-03') e left join  dept d on e.dept_id = d.id;
