#1.查询员工的姓名、年龄、职位、部门信息 (隐式内连接)
#表:emp,dept
#连接条件:emp.dept_id = dept.id

select e.name , e.age , e.job , d.name from emp e , dept d where e.dept_id = d.id;

#2、查询年龄小于30岁的员工的姓名、年龄、职位、部门信息 (显式内连接)
#表:emp , dept
#连接条件:emp,dept_id = dept.id

select e.name , e.age , e.job , d.name from emp e inner  join dept d on e.dept_id = d.id < 30;

#3.查询拥有员工的部门ID、部门名称
#表:emp,dept
#连接条件:emp.dept_id = dept.id

select d.id , d.name from emp e , dept d where  e.dept_id = d.id;

#4、查询所有年龄大于40岁的员工、及其归属的部门名称; 如果员工没有分配部门,也需要展示出来
#表：emp , dept
#连接条件: emp.dept_id = dept.id
#外连接

select * from emp e left join  dept d on e.dept_id = d.id where  e.age > 40;

#5.查询所有员工的工资等级
#表：emp , sqlgrade
#连接条件：emp.salary >= salgrade.losal and emp.salary <= salgrade.hisal

select e.* , s.grade from emp e , salgrade s where  e.salary >= s.losal and e.salary <= s.hisal;

select e.* , s.grade from emp e , salgrade s where  e.salary between s.losal and s.hisal;
